-- helpers
local function root() local c=LP.Character; return c and c:FindFirstChild("HumanoidRootPart") end
local function hum()  local c=LP.Character; return c and c:FindFirstChildOfClass("Humanoid") end
local function alive() local h=hum(); return h and h.Health > 0 end
local function isMarine() return LP.Team and LP.Team.Name == "Marines" end

local noclip = false
RunService.Stepped:Connect(function()
	if noclip and LP.Character then
		for _, p in ipairs(LP.Character:GetDescendants()) do
			if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
		end
	end
end)

local function travel(cf, timeout)
	if not cf then return end
	local r = root(); if not r then return end
	local d = (cf.Position - r.Position).Magnitude
	local tw = TweenService:Create(r, TweenInfo.new(math.clamp(d/320, .15, 3), Enum.EasingStyle.Linear), {CFrame=cf})
	tw:Play()
	local t0 = os.clock()
	repeat task.wait() r = root() until not r or (r.Position-cf.Position).Magnitude < 18 or os.clock()-t0 > (timeout or 6)
	if tw.PlaybackState == Enum.PlaybackState.Playing then tw:Cancel() end
	r = root(); if r then r.CFrame = cf end
end

local function getEnemy(name, alt)
	for _, e in ipairs(Enemies:GetChildren()) do
		local h = e:FindFirstChildOfClass("Humanoid")
		if h and h.Health > 0 and e:FindFirstChild("HumanoidRootPart") and (e.Name == name or (alt and e.Name == alt)) then
			return e
		end
	end
end

local function attack(target, keep)
	local h = target:FindFirstChildOfClass("Humanoid")
	while target.Parent and h and h.Health > 0 and keep() do
		local r = root(); local hrp = target:FindFirstChild("HumanoidRootPart")
		if not r or not hrp then break end
		r.CFrame = hrp.CFrame * CFrame.new(0, V.Height, V.Dist)
		if V.Bring then hrp.CFrame = CFrame.new(r.Position) end
		local tool = LP.Character:FindFirstChildOfClass("Tool")
		if tool then
			local lcr = tool:FindFirstChild("LeftClickRemote")
			if lcr then lcr:FireServer(Vector3.new(.01,-500,.01), 1, true); lcr:FireServer(false)
			else VIM:SendMouseButtonEvent(0,0,0,true,game,1); VIM:SendMouseButtonEvent(0,0,0,false,game,1) end
		end
		if V.FastAttack then
			for _, k in ipairs({"Z","X"}) do
				VIM:SendKeyEvent(true,k,false,game); VIM:SendKeyEvent(false,k,false,game)
			end
		end
		task.wait(.08)
	end
end

local function questGui()
	local main = LP.PlayerGui:FindFirstChild("Main")
	return main and main:FindFirstChild("Quest")
end
local function questOpen() local q = questGui(); return q and q.Visible end
local function questTitle()
	local ok, txt = pcall(function() return questGui().Container.QuestTitle.Title.Text end)
	return ok and txt or ""
end

local function pickQuest()
	local lvl = LP.Data.Level.Value
	local best
	for _, q in ipairs(QDB[Sea]) do
		if q[10] and q[10] ~= (isMarine() and "Marines" or "Pirates") then continue end
		if q[1] <= lvl then best = q end
	end
	return best or QDB[Sea][1]
end

local function enter(q)
	if q[9] and root() then
		CommF:InvokeServer("requestEntrance", q[9])
		task.wait(.6)
	end
end

local function doQuest(q)
	if not q then return end
	if questOpen() then
		if not string.find(questTitle(), q[7]) and not (q[12] and string.find(questTitle(), q[12])) then
			CommF:InvokeServer("AbandonQuest")
			task.wait(.3)
			return
		end
		local e = getEnemy(q[7], q[12])
		if e then attack(e, function() return V.AutoFarm and questOpen() end)
		else travel(q[8]) end
	elseif V.AutoQuest then
		enter(q)
		travel(q[6])
		local r = root()
		if r and (r.Position - q[6].Position).Magnitude < 14 then
			CommF:InvokeServer("SetSpawnPoint")
			CommF:InvokeServer("StartQuest", q[3], q[4])
			task.wait(.4)
		end
	end
end

task.spawn(function()
	while task.wait(.1) do
		if V.AutoFarm and alive() then
			pcall(function() doQuest(V.Selected or pickQuest()) end)
		end
	end
end)

task.spawn(function()
	while task.wait(.15) do
		if V.KillAura and alive() then
			local r = root()
			for _, e in ipairs(Enemies:GetChildren()) do
				local h = e:FindFirstChildOfClass("Humanoid")
				local hrp = e:FindFirstChild("HumanoidRootPart")
				if h and h.Health > 0 and hrp and r and (hrp.Position - r.Position).Magnitude < 120 then
					pcall(attack, e, function() return V.KillAura end)
				end
			end
		end
	end
end)

task.spawn(function()
	while task.wait(1) do
		pcall(function()
			if V.AutoBuso and LP.Character and not LP.Character:FindFirstChild("HasBuso") then CommF:InvokeServer("Buso") end
			if V.AutoStat and LP.Data.Points.Value > 0 then CommF:InvokeServer("AddPoint", V.AutoStat, 3) end
		end)
	end
end)

task.spawn(function()
	while task.wait(1) do
		if V.AutoChest and alive() then
			pcall(function()
				local r = root()
				for _, c in ipairs(CS:GetTagged("_ChestTagged")) do
					if c:IsA("Model") and c.PrimaryPart and r and (c:GetPivot().Position - r.Position).Magnitude < 400 then
						r.CFrame = c:GetPivot()
					end
				end
			end)
		end
	end
end)

LP.Idled:Connect(function()
	if V.AntiAFK then
		VU:Button2Down(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
		task.wait(1)
		VU:Button2Up(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
	end
end)

local function dropdown(parent, title, getOptions, onPick)
	local holder = new("Frame", {Size=UDim2.new(1,0,0,26), BackgroundColor3=Color3.fromRGB(38,38,50)}, parent)
	new("UICorner", {CornerRadius=UDim.new(0,6)}, holder)
	local cur = new("TextLabel", {Size=UDim2.new(1,-20,1,0), Position=UDim2.new(0,10,0,0), BackgroundTransparency=1, Font=Enum.Font.Gotham, TextSize=11, TextColor3=Color3.fromRGB(200,200,215), TextXAlignment=Enum.TextXAlignment.Left, Text=title.."  ▾"}, holder)
	local list = new("ScrollingFrame", {Size=UDim2.new(1,0,0,140), Position=UDim2.new(0,0,1,4), BackgroundColor3=Color3.fromRGB(24,24,32), CanvasSize=UDim2.new(0,0,0,0), AutomaticCanvasSize=Enum.AutomaticSize.Y, ScrollBarThickness=3, Visible=false, ZIndex=5}, holder)
	new("UICorner", {CornerRadius=UDim.new(0,6)}, list)
	local function build()
		for _, c in ipairs(list:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
		for _, opt in ipairs(getOptions()) do
			local o = new("TextButton", {Size=UDim2.new(1,0,0,24), BackgroundTransparency=1, Font=Enum.Font.Gotham, TextSize=11, TextColor3=Color3.fromRGB(210,210,225), TextXAlignment=Enum.TextXAlignment.Left, Text="  "..opt, ZIndex=6}, list)
			o.MouseButton1Click:Connect(function()
				cur.Text = opt
				list.Visible = false
				onPick(opt)
			end)
		end
	end
	new("TextButton", {Size=UDim2.new(1,0,1,0), BackgroundTransparency=1, Text="", ZIndex=4}, holder).MouseButton1Click:Connect(function()
		build(); list.Visible = not list.Visible
	end)
	return holder
end
